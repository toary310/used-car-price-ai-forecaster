# ======= 中古車価格予測アプリの依存パッケージ =======
# このファイルには、このアプリケーションが動作するために必要なすべてのPythonパッケージが記載されています。
# `pip install -r requirements.txt` コマンドでこれらのパッケージをインストールできます。

# FastAPI - 高速なWebアプリケーションフレームワーク
# RESTful APIを簡単に作成できる現代的なフレームワークで、自動ドキュメント生成機能も備えている
fastapi

# Uvicorn - ASGIサーバー
# FastAPIアプリケーションを実行するための高速なASGI対応Webサーバー
# [standard]はログやその他の拡張機能が含まれる完全版
uvicorn[standard]

# Pandas - データ分析・操作ライブラリ
# 表形式データの処理に優れた、Pythonデータ分析の中心的なライブラリ
pandas

# Scikit-learn - 機械学習ライブラリ
# 機械学習アルゴリズム（RandomForestなど）を提供する、Pythonで最も人気のある機械学習ライブラリ
scikit-learn

# Pydantic - データバリデーションライブラリ
# Pythonの型ヒントを使用してデータのバリデーション（検証）と設定管理を行うライブラリ
# FastAPIと組み合わせて入力データの検証に使用される
pydantic

# Joblib - Pythonオブジェクトの保存・読み込みライブラリ
# 特に大きな数値データ（機械学習モデルなど）の効率的な保存と読み込みに最適化されている
joblib  # scikit-learnモデルの保存/読み込みに推奨

# Python-dotenv - 環境変数管理ライブラリ
# .envファイルから環境変数を読み込むためのライブラリ（設定情報の管理に便利）
python-dotenv  # .env ファイル読み込み用 (任意)

# ===== 追加で検討できるパッケージ（現在は使用していない） =====
# numpy  # 数値計算の基盤ライブラリ（pandasやscikit-learnが自動的にインストール）
# matplotlib  # グラフ描画ライブラリ（結果の可視化に使用）
# xgboost  # 勾配ブースティングアルゴリズムの実装（高性能な機械学習モデル）
# lightgbm  # もう一つの高速な勾配ブースティングフレームワーク
